# 移动端附件切换单据缓存回归测试

关联：[[../变更记录/BUG-20260821-001-mobile-attachment-stale-cache]]

## 使用说明

以下测试代码曾在 amend 前的临时提交 `b19495448725d1bd383d50e75b0a0f07cb03c5c8` 上执行，6/6 通过；该提交已被核心代码提交 `c6cffd9a84410b4648d6b31bcb8e7210ed3b5583` 取代。为保持组件库提交只包含核心业务代码，测试源码不进入 Git；需要回归时，将两个代码块分别临时放入 `tests/udp-mobile-ui/attachment-value-change.test.tsx` 和 `tests/udp-mobile-ui/jest.config.js`，然后执行：

```powershell
node ./node_modules/jest/bin/jest.js --config ./tests/udp-mobile-ui/jest.config.js --runInBand --forceExit
```

## 覆盖场景

1. 同一组件实例从单据 A 切换到单据 B，应从 2800.pdf 更新为 4810.pdf。
2. 全新组件实例直接打开单据 B，应显示 4810.pdf。
3. 从有附件单据切换到空单据，应清空旧附件列表。
4. 相同附件绑定仅展示值结构变化时，不应重复初始化。
5. 单据 A 的旧请求晚于单据 B 返回时，不得覆盖单据 B。
6. 列附件仅数量变化时，应保留 session 且不重复初始化。

```tsx
import React, { act } from 'react';
import { createRoot, Root } from 'react-dom/client';

(globalThis as any).IS_REACT_ACT_ENVIRONMENT = true;

jest.mock('ahooks', () => ({
  useCreation: (factory: () => unknown, deps: unknown[]) => require('react').useMemo(factory, deps),
  useMemoizedFn: (fn: (...args: any[]) => any) => fn,

    return [state, setState];
  },
  useUpdateEffect: (effect: React.EffectCallback, deps: unknown[]) => {
    const mounted = require('react').useRef(false);
    require('react').useEffect(() => {
      if (mounted.current) return effect();
      mounted.current = true;
    }, deps);
  }
}));

jest.mock('antd', () => ({
  Col: ({ children }: any) => <div>{children}</div>,
  Row: ({ children }: any) => <div>{children}</div>,
  Skeleton: ({ children }: any) => <div>{children}</div>
}));
jest.mock('antd-mobile', () => ({ Toast: { show: jest.fn() } }));

jest.mock('../../packages/@newgrand/udp-mobile-ui/src/util', () => ({
  core: {
    uniqueId: (prefix: string) => `${prefix}-fixed`,
    getBusType: () => 'test-bus-type',
    getUser: () => ({ userID: 'user-1', orgID: 'org-1' }),
    getRoot: () => '',
    getHttpUrl: (url: string) => url,
    request: {
      body: jest.fn(async ({ data }: any) => ({
        data: Object.fromEntries(data.asrFids.map((fid: string) => [fid, `/preview/${fid}.pdf`]))
      }))
    },
    alert: jest.fn()
  },
  getRegisterComponent: () => undefined,
  NG: { getQueryValue: (key?: string) => (key ? '' : {}) }
}));

jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/api/service', () => ({
  initFile: jest.fn()
}));

const createInitResponse = (value?: string) => ({
  success: 'true',
  asrSessionGuid: `server-session-${value}`,
  fileList: value
    ? [value === 'doc-b' ? { AsrFid: 'fid-b', AsrName: '4810.pdf' } : { AsrFid: 'fid-a', AsrName: '2800.pdf' }]
    : []
});
jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/attachment-list', () => ({
  AttachmentList: ({ fileList }: any) => (
    <div data-testid="file-list">{fileList.map((file: any) => file.AsrName).join(',')}</div>
  )
}));
jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/attachment-preview', () => ({
  AttachmentPreview: () => null
}));
jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/attachment-type-list', () => ({
  AttachmentTypeList: () => null
}));
jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/attachment-upload', () => ({
  AttachmentUpload: () => null
}));
jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/const', () => ({ MOBILE_PLATFORMS: [] }));

  uploadAttachment: jest.fn()
}));
jest.mock('../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/util', () => ({
  getAttachUrl: (url?: string) => url || '',
  getGuidInValue: (value: string) => value.split(/[;,]|@@/)[0],
  getRealCode: (value: string, separator: string) => value.split(separator)[1],
  getUserToken: () => 'token-1'
}));

import { initFile } from '../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/api/service';
import { Attachment } from '../../packages/@newgrand/udp-mobile-ui/src/component/Attachment/attachment';

const mockInitFile = initFile as jest.Mock;

const flush = async () => {
  await act(async () => {
    await Promise.resolve();
    await Promise.resolve();
  });
};

describe('Attachment record changes', () => {
  let container: HTMLDivElement;
  let root: Root;

  beforeEach(() => {
    mockInitFile.mockReset();
    mockInitFile.mockImplementation(async ({ value }: { value?: string }) => createInitResponse(value));
    container = document.createElement('div');
    document.body.appendChild(container);
    root = createRoot(container);
  });

  afterEach(() => {
    act(() => root.unmount());
    container.remove();
  });

  it('reloads the attachment list when the mounted component switches from document A to B', async () => {
    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-a,doc-a" />);
    });
    await flush();
    expect(container.querySelector('[data-testid="file-list"]')?.textContent).toBe('2800.pdf');

    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-b,doc-b" />);
    });
    await flush();

    expect(mockInitFile).toHaveBeenCalledWith(expect.objectContaining({ value: 'doc-b' }));
    expect(container.querySelector('[data-testid="file-list"]')?.textContent).toBe('4810.pdf');
  });

  it('loads document B correctly in a fresh component instance', async () => {
    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-b,doc-b" />);
    });
    await flush();

    expect(mockInitFile).toHaveBeenCalledWith(expect.objectContaining({ value: 'doc-b' }));
    expect(container.querySelector('[data-testid="file-list"]')?.textContent).toBe('4810.pdf');
  });

  it('clears the previous list when the component switches to an empty document', async () => {
    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-b,doc-b" />);
    });
    await flush();

    await act(async () => {
      root.render(<Attachment asrtable="bill" value="" />);
    });
    await flush();

    expect(mockInitFile).toHaveBeenLastCalledWith(expect.objectContaining({ value: '' }));
    expect(container.querySelector('[data-testid="file-list"]')?.textContent).toBe('');
  });

  it('initializes an existing document only once', async () => {
    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-a,doc-a" />);
    });
    await flush();

    await act(async () => {
      root.render(<Attachment asrtable="bill" value={{ value: 'session-a,doc-a', label: [] }} />);
    });
    await flush();

    expect(mockInitFile).toHaveBeenCalledTimes(1);
    expect(mockInitFile).toHaveBeenLastCalledWith(expect.objectContaining({ value: 'doc-a' }));
  });

  it('keeps document B when the older document A request finishes last', async () => {
    let resolveDocumentA: (value: ReturnType<typeof createInitResponse>) => void = () => undefined;
    mockInitFile.mockImplementation(({ value }: { value?: string }) => {
      if (value === 'doc-a') {
        return new Promise((resolve) => {
          resolveDocumentA = resolve;
        });
      }
      return Promise.resolve(createInitResponse(value));
    });

    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-a,doc-a" />);
    });
    await act(async () => {
      root.render(<Attachment asrtable="bill" value="session-b,doc-b" />);
    });
    await flush();

    await act(async () => {
      resolveDocumentA(createInitResponse('doc-a'));
      await Promise.resolve();
    });
    await flush();

    expect(container.querySelector('[data-testid="file-list"]')?.textContent).toBe('4810.pdf');
  });

  it('keeps the column attachment session without reloading when only its count changes', async () => {
    await act(async () => {
      root.render(<Attachment asrtable="system" colAttach="1" value="column-session@@1" />);
    });
    await flush();

    await act(async () => {
      root.render(<Attachment asrtable="system" colAttach="1" value="column-session@@2" />);
    });
    await flush();

    expect(mockInitFile).toHaveBeenCalledTimes(1);
    expect(mockInitFile).toHaveBeenLastCalledWith(
      expect.objectContaining({ sessionGuid: 'column-session', value: 'column-session' })
    );
  });
});
```
